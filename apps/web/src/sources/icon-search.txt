  #setItems() {
    this.#ul.innerHTML = "";
    const result = this.#searchResult.map((item) => {
      let image = item.image;
      const idValidUrl = (() => {
        try {
          new URL(item.image);
          return true;
        } catch (e) {
          console.error(e);
          return false;
        }
      })();
      if (!idValidUrl) {
        return;
      }
      const imageUrl = new URL(item.image);
      if (imageUrl.hostname.includes("gyazo.com")) {
        image =
          imageUrl.origin + "/" + imageUrl.pathname.substring(1).split("/")[0] + "/max_size/200";
      }
      const li = document.createElement("li");
      li.addEventListener("click", () => {
        this.dispatchOnSelect(item);
        this.toggle();
      });
      li.classList.add("flex", "items-center", "justify-between", "gap-2", "pl-1", "pr-2");
      li.innerHTML = `
        <div role="button" class="flex items-center gap-2 flex-grow hover:bg-neutral-200/50 rounded-md px-2 py-1">
          <span>
            <img src="${image}" width="30px" height="30px" />
          </span>
          <span class="text-neutral-400"> [${item.title}.icon] </span>
        </div>
        <a target="_blank" href="${item.path}" class="text-blue-400 text-sm">Link</a>`;
      return li;
    }, "");
    this.#ul.append(...result);
    const itemsWrapper = this.shadowRoot.querySelector("#itemsWrapper");
    itemsWrapper.innerHTML = "";
    itemsWrapper.append(this.#ul);
  }

  toggle() {
    const isOpen = this.shadowRoot.querySelector("dialog").open;
    this.shadowRoot.querySelector("dialog").open = !isOpen;
    if (!isOpen) {
      this.shadowRoot.querySelector("input").focus();
    }
  }

  connectedCallback() {
    // this.addEventListener("click", this.handleClick);
  }

  handleEnter(event) {
    if (event.key !== "Enter") return;
    const selected = this.#searchResult[0];
    if (selected) {
      this.dispatchOnSelect(selected);
      this.toggle();
    }
  }

  handleInput(event) {
    const value = event.target.value;
    this.#searchResult = this.#icons.filter((icon) => icon.titleLc.includes(value.toLowerCase()));
    this.#setItems();
    this.dispatchOnInput(value);
  }

  async fetchIcons() {
    if (this.#icons.length) return;
    const relatedPages = (await (await fetch(`/api/pages/${this.#projectName}/icon/`)).json())
      .relatedPages.links1hop;
    this.#icons = [];
    relatedPages.forEach(({ title, titleLc, image }) => {
      if (image == null) return;
      this.#icons.push({
        title,
        titleLc,
        path: `${location.origin}/${this.#projectName}/${encodeURIComponent(title)}`,
        image,
      });
    });
    this.#searchResult = this.#icons;
    this.#setItems();
  }

  registerEvent() {
    this.shadowRoot.querySelector("input").addEventListener("input", this.handleInput.bind(this));
    this.shadowRoot.querySelector("input").addEventListener("keydown", this.handleEnter.bind(this));
    this.shadowRoot.querySelector("dialog").addEventListener("click", (e) => {
      // 子供要素にイベントを伝播させない
      e.stopPropagation();
      e.preventDefault();
      // クリックした要素が#wrapperのの中にある場合はダイアログを閉じない
      if (e.target.closest("#wrapper")) return;
      this.toggle();
    });
  }

  dispatchOnInput(value) {
    const event = new CustomEvent("input", {
      bubbles: true,
      composed: true,
      detail: { value },
    });
    this.dispatchEvent(event);
  }

  dispatchOnSelect(value) {
    const event = new CustomEvent("select", {
      bubbles: true,
      composed: true,
      detail: { value },
    });
    this.dispatchEvent(event);
  }
}

  // copy to clipboard
  navigator.clipboard.writeText(`[${e.detail.value.title}.icon]`);
  document.querySelector(".editor").focus();
  document.execCommand("insertText", null, `[${e.detail.value.title}.icon]`);
});
iconSearch.fetchIcons();
window.document.addEventListener("keydown", (e) => {
  if (e.ctrlKey && e.key === "l") {
    e.preventDefault();
    iconSearch.toggle();
  }
});
